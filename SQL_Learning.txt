//Joins
The JOIN clause is used to combine rows from two or more tables, based on a related column between them.
//Inner Join-----> Will return the matching records from both tables
Select * from employee as e
inner join department as d
ON e.empID=d.empID;

//Left Join--> Will return entire data from left side table and matching records from right side table
Select * from employee as e
Left Join department as d
ON e.empID=d.empID;

//Right Join  --> Will return entire data from right side table and matching records from the lft side table
Select * from employee as e
Right join department as d
ON e.empID=d.empID;

//Full Outer Join
Select * from employee as e
Left Join department as d
ON e.empID=d.empID;
Union
Select * from employee as e
Right join department as d
ON e.empID=d.empID;
